cmake_minimum_required(VERSION 3.10)
find_package(Threads REQUIRED)
add_library(sbmusic_sqlite STATIC third_party/sqlite/sqlite3.c)
target_compile_definitions(sbmusic_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_ENABLE_DBSTAT_VTAB SQLITE_DQS=0 _FILE_OFFSET_BITS=64)
target_compile_options(sbmusic_sqlite PRIVATE -Os)
target_link_libraries(sbmusic_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(sbmusic STATIC music_library.c ../common/music_contracts.c)
target_include_directories(sbmusic PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(sbmusic PRIVATE -Os -Wall -Wextra -Wpedantic)
target_compile_definitions(sbmusic PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(sbmusic PUBLIC sbmusic_sqlite Threads::Threads)
add_library(sbmusic_storage STATIC music_storage.c music_storage_linux.c music_storage_library.c music_storage_adapter.c)
target_compile_options(sbmusic_storage PRIVATE -Os -Wall -Wextra -Wpedantic)
target_compile_definitions(sbmusic_storage PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(sbmusic_storage PUBLIC sbmusic)
add_library(sbmusic_metadata STATIC music_metadata.c)
target_include_directories(sbmusic_metadata PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(sbmusic_metadata PRIVATE -Os -Wall -Wextra -Wpedantic)
target_compile_definitions(sbmusic_metadata PRIVATE _FILE_OFFSET_BITS=64)
add_executable(sbmusic_metadata_helper music_metadata_helper.c)
target_link_libraries(sbmusic_metadata_helper PRIVATE sbmusic_metadata)
add_library(sbmusic_scanner STATIC music_scanner.c)
target_compile_options(sbmusic_scanner PRIVATE -Os -Wall -Wextra -Wpedantic)
target_compile_definitions(sbmusic_scanner PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(sbmusic_scanner PUBLIC sbmusic_storage)

add_library(sbmusic_async STATIC music_async.c)
target_compile_options(sbmusic_async PRIVATE -Os -Wall -Wextra -Wpedantic)
target_link_libraries(sbmusic_async PUBLIC sbmusic)

add_library(sbmusic_snapshot STATIC music_snapshot.c music_snapshot_view.c)
target_compile_options(sbmusic_snapshot PRIVATE -Os -Wall -Wextra -Wpedantic)
target_compile_definitions(sbmusic_snapshot PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(sbmusic_snapshot PUBLIC sbmusic_storage)
add_executable(sbmusic_library_tool music_library_tool.c)
target_compile_options(sbmusic_library_tool PRIVATE -Os -Wall -Wextra -Wpedantic)
target_compile_definitions(sbmusic_library_tool PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(sbmusic_library_tool PRIVATE sbmusic_snapshot sbmusic_scanner)
