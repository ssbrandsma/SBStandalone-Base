cmake_minimum_required(VERSION 3.10)
project(sbwebserver C)
add_executable(sbwebserver src/main.c src/http_server.c src/auth.c src/system_info.c src/settings.c src/applets.c src/repository.c src/repository_sources.c src/repository_xml.c third_party/yxml/yxml.c src/catalog_sync.c src/radio_stations.c src/remote.c src/remote_framebuffer.c src/logs.c src/monitor.c src/image_render.c src/firmware.c src/firmware_image.c src/backup.c ../sbbase/src/alarms.c ../sbbase/src/nowplaying.c third_party/mongoose.c)
target_include_directories(sbwebserver PRIVATE src third_party third_party/yxml ../sbbase/include)
target_compile_definitions(sbwebserver PRIVATE _GNU_SOURCE _POSIX_C_SOURCE=200809L MG_ENABLE_EPOLL=0 MG_ENABLE_POLL=1 MG_ENABLE_DIRLIST=0 MG_ENABLE_SSI=0)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
target_link_libraries(sbwebserver PRIVATE Threads::Threads ZLIB::ZLIB)

endif()
