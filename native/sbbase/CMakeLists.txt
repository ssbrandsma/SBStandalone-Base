cmake_minimum_required(VERSION 3.10)
project(sbbase C)
find_package(Threads REQUIRED)
add_executable(sbbase ../sblifecycle/tinysc.c src/ownership_bridge.c ../common/upnp_ipc.c ../common/music_resource_ipc.c src/main.c src/protocol.c src/nowplaying.c src/alarms.c src/sleep.c src/lms_command.c src/output_queue.c src/slim_controller.c ../common/music_contracts.c)
target_link_libraries(sbbase PRIVATE Threads::Threads)
target_include_directories(sbbase PRIVATE include)
target_compile_definitions(sbbase PRIVATE _POSIX_C_SOURCE=200809L)
if(NOT MSVC)
  target_compile_options(sbbase PRIVATE -Os -Wall -Wextra -Wpedantic)
endif()
install(TARGETS sbbase RUNTIME DESTINATION bin)

option(SB_ENABLE_MUSIC "Reserved Standalone Music activation (unavailable in WP0)" OFF)
if(SB_ENABLE_MUSIC)
  message(FATAL_ERROR "Standalone Music runtime is not implemented in WP0")
endif()

# Isolated controller tests link the same framing/contracts; no runtime construction.
add_library(sbtransport STATIC src/slim_controller.c src/output_queue.c src/protocol.c ../common/music_contracts.c)
target_include_directories(sbtransport PUBLIC include)
if(NOT MSVC)
  target_compile_options(sbtransport PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(sbmusic_browse STATIC src/music_browse.c)
target_include_directories(sbmusic_browse PUBLIC include)
target_compile_options(sbmusic_browse PRIVATE -Os -Wall -Wextra -Wpedantic)
target_link_libraries(sbmusic_browse PUBLIC sbmusic_async)
add_library(sbmusic_queue STATIC src/music_queue.c)
target_include_directories(sbmusic_queue PUBLIC include)
target_compile_options(sbmusic_queue PRIVATE -Os -Wall -Wextra -Wpedantic)
add_library(sbmusic_load STATIC src/music_load.c)
target_include_directories(sbmusic_load PUBLIC include)
target_compile_options(sbmusic_load PRIVATE -Os -Wall -Wextra -Wpedantic)
target_link_libraries(sbmusic_load PUBLIC sbmusic_queue sbmusic_async)
add_library(sbmusic_playback STATIC src/music_playback.c src/music_file.c
  ../common/music_resource_ipc.c ../common/music_http.c ../common/remote_http.c)
target_include_directories(sbmusic_playback PUBLIC include)
target_compile_definitions(sbmusic_playback PRIVATE _FILE_OFFSET_BITS=64)
target_compile_options(sbmusic_playback PRIVATE -Os -Wall -Wextra -Wpedantic)
target_link_libraries(sbmusic_playback PUBLIC sbmusic_load sbmusic_storage sbtransport)
if(BUILD_TESTING)
  get_target_property(SB_BASE_BROWSE_TEST_SOURCES sbbase SOURCES)
  add_executable(sbbase_browse_test ${SB_BASE_BROWSE_TEST_SOURCES})
  target_include_directories(sbbase_browse_test PRIVATE include)
  target_compile_definitions(sbbase_browse_test PRIVATE _GNU_SOURCE SB_MUSIC_BROWSE_TESTING=1)
  target_compile_options(sbbase_browse_test PRIVATE -Os -Wall -Wextra -Wpedantic)
  target_link_libraries(sbbase_browse_test PRIVATE sbmusic_browse Threads::Threads)
  add_executable(sbbase_music_test ${SB_BASE_BROWSE_TEST_SOURCES})
  target_include_directories(sbbase_music_test PRIVATE include)
  target_compile_definitions(sbbase_music_test PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64
    SB_MUSIC_BROWSE_TESTING=1 SB_MUSIC_PLAYBACK_TESTING=1)
  target_compile_options(sbbase_music_test PRIVATE -Os -Wall -Wextra -Wpedantic)
  target_link_libraries(sbbase_music_test PRIVATE sbmusic_playback sbmusic_browse Threads::Threads)
endif()

option(SB_BUILD_MUSIC_SESSION "Build explicitly activated production Music session binaries" OFF)
if(SB_BUILD_MUSIC_SESSION)
 get_target_property(SB_BASE_RUNTIME_SOURCES sbbase SOURCES)
 add_executable(sbbase_music ${SB_BASE_RUNTIME_SOURCES})
 target_include_directories(sbbase_music PRIVATE include)
 target_compile_definitions(sbbase_music PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64 SB_MUSIC_RUNTIME=1)
 target_compile_options(sbbase_music PRIVATE -Os -Wall -Wextra -Wpedantic)
 target_link_libraries(sbbase_music PRIVATE sbmusic_playback sbmusic_browse sbmusic_scanner sbmusic_snapshot Threads::Threads)
 if(UNIX)
  # Host-capable production runtime: identical bootstrap/rendezvous/IPC/
  # ownership paths; only the Fab4 hardware identity check is relaxed so the
  # pairing test can run unprivileged under a user namespace.
  add_executable(sbbase_music_host_test ${SB_BASE_RUNTIME_SOURCES})
  target_include_directories(sbbase_music_host_test PRIVATE include)
  target_compile_definitions(sbbase_music_host_test PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64 SB_MUSIC_RUNTIME=1 SB_TINYSC_TEST=1)
  target_compile_options(sbbase_music_host_test PRIVATE -Os -Wall -Wextra -Wpedantic)
  target_link_libraries(sbbase_music_host_test PRIVATE sbmusic_playback sbmusic_browse sbmusic_scanner sbmusic_snapshot Threads::Threads)
 endif()
endif()
