cmake_minimum_required(VERSION 3.14)
project(sbupnp VERSION 0.3.1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(SB_UPNP_BASE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SB_UPNP_PUPNP_REVISION fac0494691434c35243131f66a0b27ea0c29f700)
find_package(Git REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include(FetchContent)
set(SB_UPNP_DEPENDENCY "${SB_UPNP_BASE}/.downloads/pupnp")
# Never discard local dependency edits or silently accept a different revision.
function(sbupnp_check_dependency)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${SB_UPNP_DEPENDENCY}" rev-parse HEAD
    OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result)
  execute_process(COMMAND "${GIT_EXECUTABLE}" --no-optional-locks -C "${SB_UPNP_DEPENDENCY}" status --porcelain
    OUTPUT_VARIABLE changes OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE status)
  if(NOT result EQUAL 0 OR NOT status EQUAL 0 OR NOT revision STREQUAL SB_UPNP_PUPNP_REVISION OR changes)
    message(FATAL_ERROR "pupnp must be clean at ${SB_UPNP_PUPNP_REVISION}; existing dependency left untouched")
  endif()
endfunction()
if(EXISTS "${SB_UPNP_DEPENDENCY}")
  sbupnp_check_dependency()
endif()
FetchContent_Declare(sbupnp_pupnp SOURCE_DIR "${SB_UPNP_DEPENDENCY}"
  GIT_REPOSITORY https://github.com/pupnp/pupnp.git
  GIT_TAG ${SB_UPNP_PUPNP_REVISION} GIT_PROGRESS TRUE)
FetchContent_GetProperties(sbupnp_pupnp)
if(NOT sbupnp_pupnp_POPULATED)
  FetchContent_Populate(sbupnp_pupnp)
endif()
sbupnp_check_dependency()
set(UPNP_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(UPNP_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(UPNP_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(UPNP_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(UPNP_ENABLE_TESTING_INTEGRATION OFF CACHE BOOL "" FORCE)
set(UPNP_ENABLE_IPV6 OFF CACHE BOOL "" FORCE)
set(UPNP_ENABLE_BACKTRACE OFF CACHE BOOL "" FORCE)
set(UPNP_IXML_ENABLE_SCRIPT_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${SB_UPNP_DEPENDENCY}" "${CMAKE_CURRENT_BINARY_DIR}/pupnp" EXCLUDE_FROM_ALL)
add_library(sbupnp_ipc "${SB_UPNP_BASE}/native/common/upnp_ipc.c")
target_include_directories(sbupnp_ipc PUBLIC "${SB_UPNP_BASE}/native/common")

target_include_directories(sbupnp_core PUBLIC src PRIVATE "${SB_UPNP_DEPENDENCY}/upnp/src/inc")
target_link_libraries(sbupnp_core PUBLIC sbupnp_ipc upnp_static ixml_static)
target_link_options(sbupnp_core INTERFACE -Wl,--wrap=ixmlParseBufferEx
  -Wl,--wrap=gena_validate_delivery_urls -Wl,-u,__wrap_ixmlParseBufferEx)
target_compile_options(sbupnp_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/renderer_descriptions.h"

    "${CMAKE_CURRENT_BINARY_DIR}/renderer_descriptions.h"
  DEPENDS generate-descriptions.py VERBATIM)

target_link_libraries(sbupnp PRIVATE sbupnp_core pthread)
target_compile_options(sbupnp PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(sbupnp PRIVATE -Wl,--gc-sections)
