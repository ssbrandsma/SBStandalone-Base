cmake_minimum_required(VERSION 3.10)
project(StandaloneBase C)
set(SB_TARGET "host" CACHE STRING "Hardware target: host, baby, fab4")
set_property(CACHE SB_TARGET PROPERTY STRINGS host baby fab4)
if(NOT SB_TARGET MATCHES "^(host|baby|fab4)$")
  message(FATAL_ERROR "Unsupported SB_TARGET")
endif()
if(SB_TARGET STREQUAL "fab4")
  add_definitions(-DSB_TARGET_FAB4=1)
endif()
include(CTest)
if(NOT WIN32)
  add_subdirectory(native/sbbase)
  add_subdirectory(native/sbwebserver)
  add_subdirectory(native/sbstorage)
  add_subdirectory(native/svgpngrenderer)
endif()
option(BUILD_SBPROXY "Build the libcurl-based proxy" ON)
if(BUILD_SBPROXY AND NOT WIN32)
  add_subdirectory(native/sbproxy)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
